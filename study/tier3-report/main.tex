% =====================================================================
%  Provenance tags.
%
%  The source carries five provenance markers:
%
%    [derived here]      -- reproducible from the numbers given in the
%                           source, but NOT a RESULTS.md row.  Often carries
%                           a long bracketed note; \derivedhere[note].
%    [RESULTS <date>]    -- a measured project number whose source of truth
%                           is RESULTS.md, with script + commit + data
%                           provenance.  If a number here disagrees with
%                           RESULTS.md, RESULTS.md wins.
%    [typical]           -- representative of a model class, not measured
%                           (pre-SN core masses, explosion energies, Ni56).
%    [assumed ...]       -- an internal estimate with a retirement plan.
%    [sourced: ...]      -- a citation with the locus stated.
%    [derived there]     -- derived in an earlier tier, not re-derived here.
%
%  All must survive the conversion; check_coverage.py counts them.
% =====================================================================
\definecolor{provrule}{HTML}{6B6151}
\newcommand{\provtag}[1]{\,{\footnotesize\textcolor{provrule}{#1}}}
\NewDocumentCommand{\derivedhere}{o}{%
  \IfNoValueTF{#1}{\provtag{\textsc{[derived here]}}}%
                  {\provtag{[\textsc{derived here} --- #1]}}}
\newcommand{\resultsref}[1]{\provtag{[\textsc{results}~#1]}}
\NewDocumentCommand{\typical}{o}{%
  \IfNoValueTF{#1}{\provtag{\textsc{[typical]}}}%
                  {\provtag{[\textsc{typical}; #1]}}}
\newcommand{\assumednote}[1]{\provtag{[\textsc{assumed} --- #1]}}
\newcommand{\sourcednote}[1]{\provtag{[\textsc{sourced}: #1]}}
\NewDocumentCommand{\derivedthere}{o}{%
  \IfNoValueTF{#1}{\provtag{\textsc{[derived there]}}}%
                  {\provtag{[\textsc{derived there}, #1]}}}
